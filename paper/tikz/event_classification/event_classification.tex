% Transaction routing classification used by dexamine.
%
% dexamine derives the route from the transaction `to` address alone, matched against
% the canonical Uniswap deployments in dexamine/shared/constants.py:uniswap_address_list.
% This gives two routed buckets -- no curated third-party label data is involved.
\documentclass[border=5pt]{standalone}
\usepackage{tikz}
\usepackage{amsmath}
\usetikzlibrary{positioning, arrows.meta, calc}

\begin{document}

\begin{tikzpicture}[
    box/.style = {rectangle, draw=black, thin, align=center,
                  minimum width=3.6cm, minimum height=1.3cm, font=\large},
    stage/.style = {font=\large\bfseries},
    code/.style = {font=\normalsize\ttfamily},
    arrow/.style = {-Latex, thin},
    line/.style = {-, thin}
]

% Stage 1: transaction initiation
\node[box] (eoa) {EOA \\ (Agent)};

% Stage 2: transaction routing
\node[box, above right=0.5cm and 2.2cm of eoa] (router)
    {Uniswap Router \\ (direct) \\[2pt] \texttt{uniswap\_router}};
\node[box, below right=0.5cm and 2.2cm of eoa] (contract)
    {Other Smart Contract \\ (indirect) \\[2pt] \texttt{other\_contract}};

% Stage 3: execution
\node[box, right=2.2cm of $(router.east)!0.5!(contract.east)$] (pool)
    {Uniswap Pool \\ (v2 / v3)};

% Stage labels, aligned on one baseline
\node[stage, anchor=south] at ($(eoa.north -| eoa) + (0, 1.35cm)$) {Tx initiation};
\node[stage, anchor=south] at ($(router.north) + (0, 0.6cm)$)
    {Tx routing (\texttt{tx\_to\_type})};
\node[stage, anchor=south] at ($(pool.north -| pool) + (0, 1.35cm)$) {Event execution};

% Arrows
\draw[arrow] (eoa.east) to[out=0, in=180] (router.west);
\draw[arrow] (eoa.east) to[out=0, in=180] (contract.west);
\draw[arrow] (router.east) to[out=0, in=180] (pool.west);
\draw[arrow] (contract.east) to[out=0, in=180] (pool.west);

\end{tikzpicture}

\end{document}
