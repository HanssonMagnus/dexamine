% How dexamine fits into a research pipeline: an indexer supplies the transaction
% positions that touched a pool, and dexamine parses each of them.
\documentclass[border=5pt]{standalone}
\usepackage{tikz}
\usepackage{amsmath}
\usetikzlibrary{positioning, arrows.meta, shapes.geometric}

\begin{document}

\begin{tikzpicture}[
    node distance = 0.7cm and 1.2cm,
    software/.style = {ellipse, draw, align=center, minimum width=4cm,
                       minimum height=1.1cm, font=\large},
    output/.style = {rectangle, draw, align=center, minimum width=4cm,
                     minimum height=1.1cm, font=\large},
    arrow/.style = {->, thick, >=Stealth},
    bidirarrow/.style = {<->, thick, >=Stealth}
]

\node[output] (user) {Input: Uniswap pool address \\ \texttt{0x88e6...}};
\node[software, below=of user] (indexer) {Indexer};
\node[output, below=of indexer] (tx_positions)
    {Transaction positions \\ \{\texttt{block\_number}, \texttt{tx\_index}\}};
\node[software, below=of tx_positions] (dexamine)
    {\texttt{dexamine} \\ (parses Uniswap events)};
\node[output, below=of dexamine] (output)
    {Output: time series \\ of Uniswap events};

\node[software, right=of tx_positions, xshift=1cm] (node)
    {Ethereum node \\ (JSON-RPC)};

\draw[arrow] (user) -- (indexer);
\draw[arrow] (indexer) -- (tx_positions);
\draw[arrow] (tx_positions) -- (dexamine);
\draw[arrow] (dexamine) -- (output);

\draw[bidirarrow] (indexer) -- (node);
\draw[bidirarrow] (dexamine) -- (node);

\end{tikzpicture}

\end{document}
